<Playground
  title="With NextJS"
  desc="Example for use with `next.js`: keep the `href` and navigate with the router on click."
  scope={{ Breadcrumbs, Grid, useRouter }}
  code={`
() => {
  const router = useRouter()
  const go = (url) => (event) => {
    event.preventDefault()
    router.push(url)
  }
  return (
    <Grid.Container gap={2}>
      <Grid>
        <Breadcrumbs>
          <Breadcrumbs.Item href="/" onClick={go('/')}>Home</Breadcrumbs.Item>
          <Breadcrumbs.Item href="/settings" onClick={go('/settings')}>Settings</Breadcrumbs.Item>
          <Breadcrumbs.Item>General</Breadcrumbs.Item>
        </Breadcrumbs>
      </Grid>
    </Grid.Container>
  )
}
`}
/>

<Attributes edit="/pages/docs/components/breadcrumbs.mdx">
<Attributes.Title>Breadcrumbs.Props</Attributes.Title>

| Attribute     | Description      | Type             | Accepted values          | Default |
| ------------- | ---------------- | ---------------- | ------------------------ | ------- |
| **separator** | separator string | `string`         | -                        | `/`     |
| ...           | native props     | `HTMLAttributes` | `'id', 'className', ...` | -       |

<Attributes.Title>Breadcrumbs.Item.Props</Attributes.Title>

| Attribute    | Description        | Type                          | Accepted values          | Default |
| ------------ | ------------------ | ----------------------------- | ------------------------ | ------- |
| **href**     | link address       | `string`                      | -                        | -       |
| **nextLink** | render as a link even without `href` | `boolean`                     | -                        | `false` |
| **onClick**  | click event        | `(event: MouseEvent) => void` | -                        | -       |
| ...          | native props       | `AnchorHTMLAttributes`        | `'id', 'className', ...` | -       |
