## useLongPress

Run a callback when an element is pressed and held, with mouse or touch.

This is custom React hooks, you need to follow the <Link target="_blank" color href="https://reactjs.org/docs/hooks-rules.html">Basic Rules</Link> when you use it.

<Playground
  desc="Hold the button for one second."
  scope={{ Button, useLongPress }}
  code={`
() => {
  const [count, setCount] = React.useState(0)
  const handlers = useLongPress(() => setCount((c) => c + 1), { threshold: 1000 })
  return <Button auto scale={0.6} {...handlers}>Held {count} times</Button>
}
`}
/>

const useLongPress = (
  onLongPress: () => void,
  options?: { threshold?: number },
) => {
  onMouseDown: () => void
  onMouseUp: () => void
  onMouseLeave: () => void
  onTouchStart: () => void
  onTouchEnd: () => void
}
