## useMutationObserver

Observe changes in the attributes, children or text of an element.

This is custom React hooks, you need to follow the <Link target="_blank" color href="https://reactjs.org/docs/hooks-rules.html">Basic Rules</Link> when you use it.

<Playground
  desc="Count the children added to the list."
  scope={{ Button, Text, useMutationObserver }}
  code={`
() => {
  const [changes, setChanges] = React.useState(0)
  const [items, setItems] = React.useState(1)
  const ref = useMutationObserver(() => setChanges((c) => c + 1), { childList: true })
  return (
    <>
      <ul ref={ref}>{Array.from({ length: items }, (_, i) => <li key={i}>Item {i + 1}</li>)}</ul>
      <Button auto scale={0.6} onClick={() => setItems(items + 1)}>Add item</Button>
      <Text>Mutations: {changes}</Text>
    </>
  )
}
`}
/>
