## useFloatingWindow

Build a floating window that the user drags by a handle and that stays inside the viewport.

This is custom React hooks, you need to follow the <Link target="_blank" color href="https://reactjs.org/docs/hooks-rules.html">Basic Rules</Link> when you use it.

<Playground
  desc="Drag the window by its title, it stays inside the viewport."
  scope={{ useFloatingWindow }}
  code={`
() => {
  const { ref, handleRef, style } = useFloatingWindow({ initialPosition: { x: 24, y: 120 } })
  return (
    <div ref={ref} style={{ ...style, zIndex: 100, width: 200, border: '1px solid currentColor', background: 'Canvas' }}>
      <div ref={handleRef} style={{ padding: 8, cursor: 'move', touchAction: 'none', borderBottom: '1px solid currentColor' }}>Drag here</div>
      <div style={{ padding: 8 }}>Floating content</div>
    </div>
  )
}
`}
/>

```ts
const useFloatingWindow = <
  T extends HTMLElement = HTMLElement,
  H extends HTMLElement = HTMLElement,
>(options?: { initialPosition?: { x: number; y: number } }) => {
  ref: RefObject<T | null>
  handleRef: RefObject<H | null>
  style: CSSProperties
  dragging: boolean
}
```
