## useScroller

Know whether a scrollable row can scroll further and scroll it by a step, for carousels and tab lists.

This is custom React hooks, you need to follow the <Link target="_blank" color href="https://reactjs.org/docs/hooks-rules.html">Basic Rules</Link> when you use it.

<Playground
  desc="The buttons are disabled when the row cannot scroll further."
  scope={{ Button, useScroller }}
  code={`
() => {
  const { ref, canScrollStart, canScrollEnd, scrollBy } = useScroller(120)
  return (
    <>
      <Button auto scale={0.6} disabled={!canScrollStart} onClick={() => scrollBy('start')}>Previous</Button>
      <Button auto scale={0.6} disabled={!canScrollEnd} onClick={() => scrollBy('end')} style={{ marginLeft: 8 }}>Next</Button>
      <div ref={ref} style={{ marginTop: 8, display: 'flex', gap: 8, overflow: 'auto', width: 240 }}>
        {Array.from({ length: 10 }, (_, i) => <div key={i} style={{ flex: '0 0 80px', padding: 16, border: '1px solid currentColor' }}>{i + 1}</div>)}
      </div>
    </>
  )
}
`}
/>

```ts
const useScroller = <T extends HTMLElement = HTMLElement>(
  step?: number,
) => {
  ref: RefObject<T | null>
  canScrollStart: boolean
  canScrollEnd: boolean
  scrollBy: (direction: 'start' | 'end') => void
}
```
