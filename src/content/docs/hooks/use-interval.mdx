## useInterval

Run a function on an interval that you can start and stop.

This is custom React hooks, you need to follow the <Link target="_blank" color href="https://reactjs.org/docs/hooks-rules.html">Basic Rules</Link> when you use it.

<Playground
  desc="A counter that ticks every second until you stop it."
  scope={{ Button, Text, useInterval }}
  code={`
() => {
  const [seconds, setSeconds] = React.useState(0)
  const interval = useInterval(() => setSeconds((s) => s + 1), 1000)
  return (
    <>
      <Text h3>{seconds}</Text>
      <Button auto scale={0.6} onClick={interval.toggle}>{interval.active ? 'Stop' : 'Start'}</Button>
    </>
  )
}
`}
/>

const useInterval = (
  fn: () => void,
  interval: number,
  options?: { autoInvoke?: boolean },

  start: () => void
  stop: () => void
  toggle: () => void
  active: boolean
}
