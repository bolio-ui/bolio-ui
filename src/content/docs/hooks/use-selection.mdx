## useSelection

Select items of a list, one by one or all at once, and know which are selected.

This is custom React hooks, you need to follow the <Link target="_blank" color href="https://reactjs.org/docs/hooks-rules.html">Basic Rules</Link> when you use it.

<Playground
  desc="Toggle items or select them all."
  scope={{ Button, Text, Spacer, useSelection }}
  code={`
() => {
  const data = ['Apple', 'Banana', 'Cherry']
  const [selected, handlers] = useSelection({ data })
  return (
    <>
      {data.map((item) => (
        <Button key={item} auto scale={0.6} type={handlers.isSelected(item) ? 'success' : 'default'} onClick={() => handlers.toggle(item)} style={{ marginRight: 8 }}>{item}</Button>
      ))}
      <Spacer h={0.5} />
      <Button auto scale={0.6} onClick={handlers.selectAll}>Select all</Button>
      <Button auto scale={0.6} onClick={handlers.deselectAll} style={{ marginLeft: 8 }}>Clear</Button>
      <Text>Selected: {selected.join(', ') || 'none'}</Text>
    </>
  )
}
`}
/>

```ts
const useSelection = <T>(options: {
  data: T[]
  initialSelection?: T[]
}) => [
  T[],
  {
    select: (item: T) => void
    deselect: (item: T) => void
    toggle: (item: T) => void
    selectAll: () => void
    deselectAll: () => void
    setSelection: (items: T[]) => void
    isSelected: (item: T) => boolean
  },
]
```
