## useLocalStorage

Keep a value in localStorage, in sync across components and tabs, safe for server rendering.

This is custom React hooks, you need to follow the <Link target="_blank" color href="https://reactjs.org/docs/hooks-rules.html">Basic Rules</Link> when you use it.

<Playground
  desc="The value survives a page reload."
  scope={{ Button, Input, Spacer, useLocalStorage }}
  code={`
() => {
  const [name, setName, removeName] = useLocalStorage({ key: 'bolio-ui-docs-name', defaultValue: '' })
  return (
    <>
      <Input placeholder="Your name" value={name} onChange={(e) => setName(e.target.value)} />
      <Spacer h={0.5} />
      <Button auto scale={0.6} onClick={removeName}>Clear</Button>
    </>
  )
}
`}
/>

const useLocalStorage = <T>(options: {
  key: string
  defaultValue: T
}) => [
  T,
  (value: T | ((current: T) => T)) => void,
  () => void,
]
