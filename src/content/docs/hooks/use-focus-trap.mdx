## useFocusTrap

Keep the focus inside an element while it is active, cycling with Tab.

This is custom React hooks, you need to follow the <Link target="_blank" color href="https://reactjs.org/docs/hooks-rules.html">Basic Rules</Link> when you use it.

<Playground
  desc="Press Tab inside the box, the focus never leaves it."
  scope={{ Button, Input, Spacer, useFocusTrap }}
  code={`
() => {
  const [active, setActive] = React.useState(false)
  const ref = useFocusTrap(active)
  return (
    <>
      <Button auto scale={0.6} onClick={() => setActive(!active)}>{active ? 'Release focus' : 'Trap focus'}</Button>
      <div ref={ref} style={{ marginTop: 8, padding: 16, border: '1px dashed currentColor' }}>
        <Input placeholder="First" />
        <Spacer h={0.5} />
        <Input placeholder="Second" />
      </div>
    </>
  )
}
`}
/>
