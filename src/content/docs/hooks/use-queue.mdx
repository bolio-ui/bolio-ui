## useQueue

Show a limited number of items at once and keep the rest waiting in a queue.

This is custom React hooks, you need to follow the <Link target="_blank" color href="https://reactjs.org/docs/hooks-rules.html">Basic Rules</Link> when you use it.

<Playground
  desc="Only two items are visible, the others wait in the queue."
  scope={{ Button, Text, useQueue }}
  code={`
() => {
  const { state, queue, add, update } = useQueue({ initialValues: [1, 2, 3], limit: 2 })
  return (
    <>
      <Text>Visible: {state.join(', ')}</Text>
      <Text>Queue: {queue.join(', ') || 'empty'}</Text>
      <Button auto scale={0.6} onClick={() => add(state.length + queue.length + 1)}>Add</Button>
      <Button auto scale={0.6} onClick={() => update((items) => items.slice(1))} style={{ marginLeft: 8 }}>Remove first</Button>
    </>
  )
}
`}
/>
