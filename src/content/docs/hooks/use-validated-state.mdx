## useValidatedState

Keep a value together with its validity and the last value that was valid.

This is custom React hooks, you need to follow the <Link target="_blank" color href="https://reactjs.org/docs/hooks-rules.html">Basic Rules</Link> when you use it.

<Playground
  desc="The email is valid when it matches a simple pattern."
  scope={{ Text, Input, Spacer, useValidatedState }}
  code={`
() => {
  const [{ value, lastValidValue, valid }, setEmail] = useValidatedState('', (v) => /^\S+@\S+\.\S+$/.test(v))
  return (
    <>
      <Input placeholder="Email" value={value} onChange={(e) => setEmail(e.target.value)} />
      <Spacer h={0.5} />
      <Text>Valid: {String(valid)}, last valid: {lastValidValue || 'none'}</Text>
    </>
  )
}
`}
/>

```ts
const useValidatedState = <T>(
  initialValue: T,
  validate: (value: T) => boolean,
) => [
  { value: T; lastValidValue: T | undefined; valid: boolean },
  (value: T) => void,
]
```
