## useToast

Display an important message globally. A toast is not a component you render: call `setToast` from `useToasts` anywhere under `BolioUIProvider`, and the provider shows it. It disappears after `delay`, and stays while the pointer is over it.

<Playground
  desc="Basic usage. The toast closes after 2 seconds, or after `delay` milliseconds."
  scope={{ Button, useToasts }}
  code={`
() => {
  const { setToast } = useToasts()
  const click = () => setToast({ text: 'Draft saved.', delay: 3000 })
  return <Button scale={2/3} auto onClick={click}>Save draft</Button>
}
`}
/>

  )
}
`}

<Playground
  title="Action"
  desc="Add a button with `actions`. Its `handler` receives the click and a `cancel` function that closes the toast. Actions show only when `text` is a string."
  scope={{ Button, useToasts }}
  code={`
() => {
  const { setToast } = useToasts()
  const archive = () => {
    setToast({
      text: 'Conversation archived.',
      delay: 6000,
      actions: [{
        name: 'Undo',
        handler: (event, cancel) => {
          cancel()
          setToast({ text: 'Conversation restored.' })
        }
      }]
    })
  }
  return <Button scale={2/3} auto onClick={archive}>Archive conversation</Button>
}
`}
/>

<Playground
  title="Unique id"
  desc="Give a toast an `id` to close it later with `removeToastOneByID`, or use `removeAll` to close every toast. Setting a toast with an `id` that is still shown throws an error, so check with `findToastOneByID` first."
  scope={{ Button, useToasts, Spacer }}
  code={`
() => {
  const { setToast, findToastOneByID, removeToastOneByID, removeAll } = useToasts()
  const start = () => {
    if (findToastOneByID('upload')) return
    setToast({ id: 'upload', text: 'Uploading 3 files.', delay: 20000 })
  }
  return (
    <>
      <Button scale={2/3} auto onClick={start}>Start upload</Button>
      <Spacer inline w={.5} />
      <Button scale={2/3} auto onClick={() => removeToastOneByID('upload')}>Cancel upload</Button>
      <Spacer inline w={.5} />
      <Button scale={2/3} auto onClick={removeAll}>Clear all</Button>
    </>
  )
}
`}
/>

<Attributes edit="/content/docs/hooks/use-toast.mdx">
<Attributes.Title>useToasts</Attributes.Title>

```ts
const {
  // All Toasts currently in DOM
  toasts: Array<Toast>

  // Set a displayable Toast
  setToast: (toast: ToastInput) => void

  // Hide all Toast immediately
  removeAll: () => void

  // Query by id to get an instance of Toast
  findToastOneByID: (ident: string) => Toast | undefined

  // Remove the specified Toast by id
  removeToastOneByID: (ident: string) => void
} = useToasts(layout: ToastLayout)
```

<Attributes.Title>ToastLayout</Attributes.Title>

| Option        | Description                  | Type                              | Default       |
| ------------- | ---------------------------- | --------------------------------- | ------------- |
| **padding**   | space inside each toast      | `string`                          | `12px 16px`   |
| **margin**    | space around each toast      | `string`                          | `8px 0`       |
| **width**     | width of each toast          | `string`                          | `420px`       |
| **maxWidth**  | largest width of a toast     | `string`                          | `90vw`        |
| **maxHeight** | largest height of a toast    | `string`                          | `75px`        |
| **placement** | the pop-up position of toast | [ToastPlacement](#toastplacement) | `bottomRight` |

<Attributes.Title>ToastInput</Attributes.Title>

| Option      | Description                                  | Type                        | Default   |
| ----------- | -------------------------------------------- | --------------------------- | --------- |
| **id**      | name to find or close the toast. It must not match a toast that is still shown | `string`                    | generated |
| **text**    | content displayed in toast                   | `string`, `ReactNode`       | -         |
| **type**    | the type of toast                            | [ToastTypes](#toasttypes)   | `default` |
| **delay**   | milliseconds before it closes. `0` uses the default | `number`             | `2000`    |
| **actions** | buttons shown after the text. Only with a string `text` | [ToastAction](#toastaction)[] | -  |

<Attributes.Title>ToastPlacement</Attributes.Title>

```ts
type ToastPlacement = 'topLeft' | 'topRight' | 'bottomLeft' | 'bottomRight'
```

<Attributes.Title>ToastAction</Attributes.Title>

```ts
interface ToastAction {
  name: string
  handler: (
    event: React.MouseEvent<HTMLButtonElement>,
    cancel: () => void
  ) => void
  passive?: boolean
}
```

<Attributes.Title>ToastTypes</Attributes.Title>

```ts
type ToastTypes =
  | 'default'
  | 'primary'
  | 'secondary'
  | 'success'
  | 'warning'
  | 'error'
  | 'info'
```
