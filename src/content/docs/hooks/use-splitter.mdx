## useSplitter

Resize two panes by dragging the handle between them, in percent of the container.

This is custom React hooks, you need to follow the <Link target="_blank" color href="https://reactjs.org/docs/hooks-rules.html">Basic Rules</Link> when you use it.

<Playground
  desc="Drag the bar to resize the panes."
  scope={{ useSplitter }}
  code={`
() => {
  const { containerRef, handleRef, size } = useSplitter({ initialSize: 40 })
  return (
    <div ref={containerRef} style={{ display: 'flex', height: 100, border: '1px dashed currentColor' }}>
      <div style={{ width: size + '%', padding: 8 }}>{Math.round(size)}%</div>
      <div ref={handleRef} style={{ width: 8, cursor: 'col-resize', touchAction: 'none', background: 'currentColor', opacity: 0.3 }} />
      <div style={{ flex: 1, padding: 8 }}>Second pane</div>
    </div>
  )
}
`}
/>

const useSplitter = <

  min?: number
  max?: number
  direction?: 'horizontal' | 'vertical'

  handleRef: RefObject<H | null>
