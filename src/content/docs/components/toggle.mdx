<Playground
  title="Notification settings"
  desc="Put the label and what it does on the left and the `Toggle` on the right, and name each switch with `aria-label`."
  scope={{ Toggle, Card, Text }}
  code={`
() => {
  const [prefs, setPrefs] = React.useState({ email: true, push: false, news: false })
  const rows = [
    ['email', 'Email', 'Comments and mentions'],
    ['push', 'Push', 'Alerts on your phone'],
    ['news', 'Newsletter', 'One email a month']
  ]
  const update = (key) => (event) =>
    setPrefs((last) => ({ ...last, [key]: event.target.checked }))
  return (
    <Card bordered width="100%">
      {rows.map(([key, label, hint]) => (
        <div key={key} style={{ display: 'flex', alignItems: 'center', justifyContent: 'space-between', padding: '8px 0' }}>
          <div>
            <Text p my={0}><b>{label}</b></Text>
            <Text small type="secondary">{hint}</Text>
          </div>
          <Toggle checked={prefs[key]} onChange={update(key)} aria-label={label} />
        </div>
      ))}
    </Card>
  )
}
`}
/>

<Attributes edit="/content/docs/components/toggle.mdx">
<Attributes.Title>Toggle.Props</Attributes.Title>

| Attribute          | Description               | Type                  | Accepted values                    | Default   |
| ------------------ | ------------------------- | --------------------- | ---------------------------------- | --------- |
| **checked**        | checked or not            | `boolean`             | -                                  | -         |
| **initialChecked** | checked or not on initial | `boolean`             | -                                  | `false`   |
| **onChange**       | change event handler      | `ToggleEvent`         | -                                  | -         |
| **type**           | current type              | `ToggleTypes`         | [ToggleTypes](#toggletypes)        | `default` |
| **disabled**       | disable toggle            | `boolean`             | -                                  | `false`   |
| **icon**           | icon inside the thumb     | `ReactNode`           | -                                  | -         |
| **checkedIcon**    | icon when checked, falls back to `icon` | `ReactNode` | -                              | -         |
| ...                | native props              | `LabelHTMLAttributes` | `'from', 'name', 'className', ...` | -         |

<Attributes.Title>ToggleEvent</Attributes.Title>

```ts
interface ToggleEventTarget {
  checked: boolean
}

export interface ToggleEvent {
  target: ToggleEventTarget
  stopPropagation: () => void
  preventDefault: () => void
  nativeEvent: React.ChangeEvent
}
```
