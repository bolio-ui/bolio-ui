## RollingNumber

A number whose digits roll to the new value when it changes. It reads as plain text for assistive technology and stops rolling when the user prefers reduced motion.

<Playground
  desc="Change the value and each digit rolls to its place."
  scope={{ RollingNumber, Button, Grid, Text }}
  code={`
() => {
  const [value, setValue] = React.useState(1280)
  return (
    <Grid.Container gap={2} alignItems="center">
      <Grid>
        <Text h2 my={0}><RollingNumber value={value} /></Text>
      </Grid>
      <Grid>
        <Button auto scale={0.6} onClick={() => setValue(value + 137)}>Add 137</Button>
      </Grid>
    </Grid.Container>
  )
}
`}
/>

<Playground
  title="Format"

  scope={{ RollingNumber, Button, Grid, Text }}

() => {
  const [value, setValue] = React.useState(4820.5)
  return (
    <Grid.Container gap={2} alignItems="center">
      <Grid>
        <Text h2 my={0}>
          <RollingNumber value={value} formatOptions={{ style: 'currency', currency: 'USD' }} />
        </Text>
      </Grid>
      <Grid>
        <Button auto scale={0.6} onClick={() => setValue(value * 2)}>Double</Button>
      </Grid>
    </Grid.Container>
  )
}

/>
