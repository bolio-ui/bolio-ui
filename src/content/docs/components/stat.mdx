## Stat

Show a number with its label and trend.

<Playground
  title="General"
  desc="The basic `Stat` usage. A trend that starts with `-` is shown as a drop."
  scope={{ Stat, Grid, Card }}
  code={`
<Grid.Container gap={2}>
  <Grid xs={12} md={4}>
    <Card bordered width="100%">
      <Stat label="Revenue" value="$48,200" change="+12.4% this month" />
    </Card>
  </Grid>
  <Grid xs={12} md={4}>
    <Card bordered width="100%">
      <Stat label="Active users" value="3,842" change="+4.1% this week" />
    </Card>
  </Grid>
  <Grid xs={12} md={4}>
    <Card bordered width="100%">
      <Stat label="Churn" value="1.8%" change="-0.3% this month" />
    </Card>
  </Grid>
</Grid.Container>
`}
/>

<Playground
  title="Dashboard with a period"
  desc="Pair the `Stat` cards with a `SegmentedControl` that changes the period they show."
  scope={{ Stat, Grid, Card, SegmentedControl }}
  code={`
() => {
  const periods = {
    week: [
      ['Revenue', '$11,300', '+3.2% vs last week'],
      ['Active users', '2,914', '+1.1% vs last week'],
      ['Churn', '0.4%', '-0.1% vs last week']
    ],
    month: [
      ['Revenue', '$48,200', '+12.4% vs last month'],
      ['Active users', '3,842', '+4.1% vs last month'],
      ['Churn', '1.8%', '-0.3% vs last month']
    ],
    year: [
      ['Revenue', '$512,900', '+31.0% vs last year'],
      ['Active users', '3,842', '+58.2% vs last year'],
      ['Churn', '6.2%', '-1.4% vs last year']
    ]
  }
  const [period, setPeriod] = React.useState('month')
  return (
    <Grid.Container gap={2} direction="column">
      <Grid>
        <SegmentedControl value={period} onChange={setPeriod} aria-label="Period">
          <SegmentedControl.Item value="week">Week</SegmentedControl.Item>
          <SegmentedControl.Item value="month">Month</SegmentedControl.Item>
          <SegmentedControl.Item value="year">Year</SegmentedControl.Item>
        </SegmentedControl>
      </Grid>
      <Grid>
        <Grid.Container gap={2}>
          {periods[period].map(([label, value, change]) => (
            <Grid key={label} xs={12} md={4}>
              <Card bordered width="100%">
                <Stat label={label} value={value} change={change} />
              </Card>
            </Grid>
          ))}
        </Grid.Container>
      </Grid>
    </Grid.Container>
  )
}
`}
/>

<Attributes.Title>Stat.Props</Attributes.Title>

| Attribute  | Description                                | Type             | Accepted values | Default |
| ---------- | ------------------------------------------ | ---------------- | --------------- | ------- |
| **label**  | what the number measures                   | `ReactNode`      | -               | -       |
| **value**  | the number                                 | `ReactNode`      | -               | -       |
| **change** | trend text, a leading `-` is shown as drop | `string`         | -               | -       |
| **extra**  | chart or icon on the right                 | `ReactNode`      | -               | -       |
| ...        | native props                               | `HTMLAttributes` | -               | -       |
