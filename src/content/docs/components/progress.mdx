<Playground
  title="Upload progress"
  desc="Say what is being sent and how far it is, and change the type when it finishes."
  scope={{ Progress, Grid, Text, Button }}
  code={`
() => {
  const [value, setValue] = React.useState(0)
  const upload = () => {
    setValue(10)
    const timer = setInterval(() => {
      setValue((last) => {
        if (last >= 100) {
          clearInterval(timer)
          return 100
        }
        return last + 10
      })
    }, 300)
  }
  const label = value === 100 ? 'Upload complete' : value ? 'Uploading report.pdf, ' + value + '%' : 'report.pdf is ready to upload'
  return (
    <Grid.Container direction="column" gap={1}>
      <Grid>
        <Text small>{label}</Text>
      </Grid>
      <Grid>
        <Progress value={value} type={value === 100 ? 'success' : 'primary'} />
      </Grid>
      <Grid>
        <Button auto scale={0.75} onClick={upload} disabled={value > 0 && value < 100}>Upload</Button>
      </Grid>
    </Grid.Container>
  )
}
`}
/>

<Attributes edit="/content/docs/components/progress.mdx">
<Attributes.Title>Progress.Props</Attributes.Title>

| Attribute       | Description            | Type                        | Accepted values                 | Default   |
| --------------- | ---------------------- | --------------------------- | ------------------------------- | --------- |
| **value**       | current value          | `number`                    | -                               | 0         |
| **max**         | max value              | `number`                    | -                               | 100       |
| **fixedTop**    | fix progress to top    | `boolean`                   | -                               | `false`   |
| **fixedBottom** | fix progress to bottom | `boolean`                   | -                               | `false`   |
| **segments**    | split the bar in blocks | `number`                    | -                               | -         |
| **circular**    | show a ring instead of a bar | `boolean`              | -                               | `false`   |
| **colors**      | custom colors          | `{ [key: number]: string }` | -                               | -         |
| **type**        | predefined state types | `ProgressTypes`             | [ProgressTypes](#progresstypes) | `default` |
| ...             | native props           | `ProgressHTMLAttributes`    | `'aria-busy', ...`              | -         |

<Attributes.Title>ProgressTypes</Attributes.Title>

```ts
type ProgressTypes =
  | 'default'
  | 'primary'
  | 'secondary'
  | 'success'
  | 'warning'
  | 'error'
  | 'info'
```
