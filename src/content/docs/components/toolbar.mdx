## Toolbar

Group related controls in one tab stop and move between them with the arrow keys.

<Playground

<Playground
  title="Editor actions"
  desc="Mix buttons and chips. The text below shows which formats are on."
  scope={{ Toolbar, Button, Chip, Text }}
  code={`
() => {
  const [formats, setFormats] = React.useState(['bold'])
  const toggle = (name) =>
    setFormats((current) => current.includes(name) ? current.filter((f) => f !== name) : [...current, name])
  return (
    <>
      <Toolbar aria-label="Formatting">
        {['bold', 'italic', 'underline'].map((name) => (
          <Chip key={name} checked={formats.includes(name)} onChange={() => toggle(name)} style={{ textTransform: 'capitalize' }}>
            {name}
          </Chip>
        ))}
        <Toolbar.Separator />
        <Button auto scale={0.75} subtle onClick={() => setFormats([])}>Clear</Button>
      </Toolbar>
      <Text small type="secondary">{formats.length ? 'On: ' + formats.join(', ') : 'No format is on.'}</Text>
    </>
  )
}
`}
/>
