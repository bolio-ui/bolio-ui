## Spinner

Indicate an action running in the background.

As an indicator of load waiting, see also component [Loading](/docs/components/loading).

<Playground
  title="Loading a panel"
  desc="Show the `Spinner` where the content will appear, and swap it for the content when it arrives."
  scope={{ Spinner, Card, Button, Text, Grid }}
  code={`
() => {
  const [loading, setLoading] = React.useState(false)
  const refresh = () => {
    setLoading(true)
    setTimeout(() => setLoading(false), 1500)
  }
  return (
    <Card bordered width="100%">
      <Grid.Container justify="space-between" alignItems="center">
        <Grid>
          <Text b>Recent invoices</Text>
        </Grid>
        <Grid>
          <Button auto scale={0.75} disabled={loading} onClick={refresh}>
            Refresh
          </Button>
        </Grid>
      </Grid.Container>
      <div style={{ minHeight: 96, display: 'flex', alignItems: 'center', justifyContent: 'center' }}>
        {loading ? <Spinner scale={1.5} /> : <Text p>3 invoices are waiting for payment.</Text>}
      </div>
    </Card>
  )
}
`}
/>
